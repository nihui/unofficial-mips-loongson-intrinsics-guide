__m256i __lasx_mxror_b (__m256i a, __m256i b)

Synopsis

__m256i __lasx_mxror_b (__m256i a, __m256i b)
#include <loongson-asxintrin.h>
Instruction: mxror.b
Builtin: __builtin_lasx_mxror_b
CPU Flags: __mips_loongson_asx
